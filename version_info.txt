# PyInstaller 版本资源定义（写进 exe 的 VS_VERSIONINFO）
#
# 为什么要这个：
#   一个「未签名 + 无公司名 / 无产品名 / 无版本号」的 PE 文件，是启发式引擎
#   （Defender 的 !ml 系列检测）最容易被盯上的形态 —— 因为正常发布的商业/开源
#   软件不会长这样，长这样的多数是免杀马。
#
#   补上版本资源不解决签名问题，但能显著降低「看起来像恶意软件」的权重，
#   并且在文件「属性 → 详细信息」里能看到发布者信息。
#
# 用法：dsh-tray.spec 里 EXE(..., version=os.path.join(HERE, 'version_info.txt'))
#
# 改版本号时记得同步 filevers / prodvers / FileVersion / ProductVersion 四处。
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 1, 0),
    prodvers=(0, 1, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '080404B0',
        [StringStruct('CompanyName', 'Konmin'),
         StringStruct('FileDescription', 'Delfin - DeepSeek Harness (dsh) tray controller'),
         StringStruct('FileVersion', '0.1.1'),
         StringStruct('InternalName', 'dsh-tray'),
         StringStruct('LegalCopyright', 'MIT License - Copyright (c) 2026 Konmin'),
         StringStruct('OriginalFilename', 'dsh-tray.exe'),
         StringStruct('ProductName', 'Delfin'),
         StringStruct('ProductVersion', '0.1.1')])
    ]),
    VarFileInfo([VarStruct('Translation', [2052, 1200])])
  ]
)
